// server.js const express = require('express'); const cors = require('cors'); const path = require('path'); const api = require('./controller/api_login'); const app = express(); const PORT = 3000; // Middlewares app.use(cors()); app.use(express.json()); app.use(express.urlencoded({ extended: true })); // Servir archivos estáticos (tu HTML, CSS, JS del dashboard) app.use(express.static(path.join(__dirname, 'client'))); // Rutas API app.use('/api/auth', api); // Ruta principal para el login app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'client', 'tecnico', 'login.html')); }); // Ruta para el dashboard app.get('/dashboard', (req, res) => { res.sendFile(path.join(__dirname, 'client', 'tecnico', 'dashboard.html')); }); app.listen(PORT, () => { console.log(`🚀 Servidor en http://localhost:${PORT}`); console.log(`📡 API login: http://localhost:${PORT}/api/auth/login`); });
